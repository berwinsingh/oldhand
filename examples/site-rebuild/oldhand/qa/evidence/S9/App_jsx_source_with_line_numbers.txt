   102	function CopyButton({ value, label, text = "Copy" }) {
   103	  const [state, setState] = useState("idle");
   104	  async function copy(event) {
   105	    const button = event.currentTarget;
   106	    let ok = false;
   107	    try {
   108	      await navigator.clipboard.writeText(value);
   109	      ok = true;
   110	    } catch {
   111	      const textarea = document.createElement("textarea");
   112	      textarea.value = value;
   113	      textarea.style.position = "fixed";
   114	      textarea.style.opacity = "0";
   115	      document.body.appendChild(textarea);
   116	      textarea.select();
   117	      ok = document.execCommand("copy");
   118	      textarea.remove();
   119	      button.focus();
   120	    }
   121	    setState(ok ? "copied" : "failed");
   122	    window.setTimeout(() => setState("idle"), 1600);
   123	  }
   124	  const shown = state === "copied" ? "Copied" : state === "failed" ? "Copy failed: select the text" : text;
   125	  return (
   126	    <button className="copy" type="button" onClick={copy} aria-label={label}>
   127	      <span role="status">{shown}</span>
   128	    </button>
   129	  );
   130	}

   146	function InstallSwitcher() {
   147	  const [tab, setTab] = useState(tabFromUrl);
   148	  const current = installs.find((i) => i.key === tab);
   149	
   150	  function onKeyDown(event) {
   151	    if (event.metaKey || event.altKey || event.ctrlKey || event.shiftKey) return;
   152	    const step = event.key === "ArrowRight" ? 1 : event.key === "ArrowLeft" ? -1 : 0;
   153	    if (!step) return;
   154	    event.preventDefault();
   155	    const index = installs.findIndex((i) => i.key === tab);
   156	    const next = installs[(index + step + installs.length) % installs.length].key;
   157	    setTab(next);
   158	    document.getElementById(`tab-${next}`)?.focus();
   159	  }
   160	
   161	  return (
   162	    <div className="install" id="install">
   163	      <div className="tabs" role="tablist" aria-label="Install route" onKeyDown={onKeyDown}>
   164	        {installs.map((i) => (
   165	          <button
   166	            key={i.key}
   167	            id={`tab-${i.key}`}
   168	            role="tab"
   169	            type="button"
   170	            aria-selected={i.key === tab}
   171	            aria-controls={`panel-${i.key}`}
   172	            tabIndex={i.key === tab ? 0 : -1}
   173	            className={i.key === tab ? "is-active" : ""}
   174	            onClick={() => setTab(i.key)}
   175	          >
   176	            {i.label}
   177	          </button>
   178	        ))}
   179	      </div>
   180	      <div className="panel" role="tabpanel" id={`panel-${current.key}`} aria-labelledby={`tab-${current.key}`}>
   181	        {current.blocks.map((block, n) => <Command key={block} text={block} index={n + 1} />)}
   182	        <p className="note">{current.note}</p>
   183	        <dl className="then">
   184	          {current.then.map(([hosts, text]) => (
   185	            <div key={hosts}>
   186	              <dt>Then, in {hosts}, for example</dt>
   187	              <dd><code>{text}</code></dd>
   188	            </div>
   189	          ))}
   190	        </dl>
