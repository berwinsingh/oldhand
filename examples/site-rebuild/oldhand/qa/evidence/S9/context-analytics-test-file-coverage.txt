    14	    head,
    15	    createElement: () => ({ dataset: {} }),

    17	  };

    21	function fakeStorage(initial = {}) {
    22	  const data = { ...initial };
    23	  return { getItem: (k) => (k in data ? data[k] : null), setItem: (k, v) => { data[k] = v; }, data };
    24	}

    39	test("granted sets consent default denied before granting, then injects one script", () => {
    40	  const { win, doc } = fakeDom();
    41	  applyAnalyticsChoice("granted", win, doc);
    42	  const calls = win.dataLayer.map((args) => Array.from(args));
    43	  assert.deepEqual(calls[0], ["consent", "default", {
    44	    analytics_storage: "denied", ad_storage: "denied", ad_user_data: "denied", ad_personalization: "denied",
    45	  }]);
    46	  assert.deepEqual(calls[1], ["consent", "update", { analytics_storage: "granted" }]);
    47	  assert.equal(doc.head.children.length, 1);
    48	  assert.match(doc.head.children[0].src, new RegExp(`id=${GA_MEASUREMENT_ID}$`));
    49	  applyAnalyticsChoice("granted", win, doc);
    50	  assert.equal(doc.head.children.length, 1, "a second grant does not inject a second script");
    51	});

    53	test("denied after a load updates consent to denied", () => {
