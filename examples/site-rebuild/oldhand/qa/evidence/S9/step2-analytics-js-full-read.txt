     1	// Opt-in Google Analytics. Nothing loads until a stored "granted" choice exists.
     2	export const GA_MEASUREMENT_ID = "G-75YW6HE9FT";

     7	    const store = storage || window.localStorage;
     8	    const choice = store.getItem(ANALYTICS_CONSENT_KEY);
     9	    return choice === "granted" || choice === "denied" ? choice : null;
    10	  } catch {

    12	  }
    13	}

    23	export function loadAnalytics(win, doc) {
    24	  const w = win || window;
    25	  const d = doc || document;
    26	  w.dataLayer = w.dataLayer || [];
    27	  w.gtag =
    28	    w.gtag ||
    29	    function gtag() {
    30	      w.dataLayer.push(arguments);
    31	    };
    32	
    33	  if (d.querySelector("script[data-oldhand-analytics]")) {
    34	    w.gtag("consent", "update", { analytics_storage: "granted" });
    35	    return;
    36	  }
    37	
    38	  w.gtag("consent", "default", {
    39	    analytics_storage: "denied",
    40	    ad_storage: "denied",
    41	    ad_user_data: "denied",
    42	    ad_personalization: "denied",
    43	  });
    44	  w.gtag("consent", "update", { analytics_storage: "granted" });
    45	  w.gtag("js", new Date());

    47

    52	  d.head.appendChild(script);

    55	// Apply a choice: only "granted" loads anything; "denied" tells an already
    56	// loaded gtag to stop; null (no choice yet) does nothing.
    57	export function applyAnalyticsChoice(choice, win, doc) {
    58	  const w = win || window;
    59	  if (choice === "granted") loadAnalytics(w, doc);
    60	  else if (choice === "denied" && typeof w.gtag === "function") {
